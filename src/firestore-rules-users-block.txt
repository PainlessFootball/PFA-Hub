rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function getRole() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    function isAdmin()     { return getRole() == 'admin'; }
    function isMod()       { return getRole() == 'moderator' || isAdmin(); }
    function isUser()      { return request.auth != null; }
    function isSelf(uid)   { return request.auth != null && request.auth.uid == uid; }

    match /users/{uid} {
      allow read: if request.auth != null;
      allow create: if isSelf(uid)
        && request.resource.data.role ==
           (request.resource.data.email in ['jdshort99@gmail.com', 'painlessfootball@gmail.com'] ? 'admin' : 'user')
        && request.resource.data.approved ==
           (request.resource.data.email in ['jdshort99@gmail.com', 'painlessfootball@gmail.com'])
        && request.resource.data.pendingApproval == false;
      allow update: if isSelf(uid)
        && request.resource.data.role == resource.data.role
        && request.resource.data.approved == resource.data.approved
        && request.resource.data.everApproved == resource.data.everApproved
        && request.resource.data.rejected == resource.data.rejected
        && request.resource.data.email == resource.data.email;
      allow update: if isAdmin();
      allow delete: if isSelf(uid) || isAdmin();
    }
    match /chat/{msg} {
      allow read:   if isUser();
      allow create: if isUser();
      allow delete: if isMod();
    }
    match /news/{item} {
      allow read:   if isUser();
      allow write:  if isMod();
    }
    match /applications/{app} {
      allow create: if isUser();
      allow read:   if isMod();
      allow update: if isAdmin();
    }
    match /settings/{doc} {
      allow read:  if isUser();
      allow write: if isAdmin();
    }
    match /weeklyResults/{doc} {
      allow read:            if isUser();
      allow create, update:  if isUser();
      allow delete:          if isAdmin();
    }
    match /club300Live/{doc} {
      allow read:            if isUser();
      allow create, update:  if isUser();
      allow delete:          if isAdmin();
    }

    // NEW — Tournament frozen-seed snapshot. Anyone can read; any signed-in
    // coach's browser can create it the FIRST time (whoever happens to load
    // the Tournament tab once Week 8 starts), but once it exists it can
    // never be overwritten or deleted through the client — same
    // write-once/read-thereafter shape as the seeding logic itself needs.
    match /tournamentSeeds/{year} {
      allow read: if true;
      allow create: if isUser()
        && !exists(/databases/$(database)/documents/tournamentSeeds/$(year));
      allow update, delete: if false;
    }
  }
}
